    1. Create `src/frontend/src/stores/chat.ts` — Pinia chat store:
       ```typescript
       import { defineStore } from 'pinia'
       import { ref, computed } from 'vue'

       export interface ChatMessage {
         id: string
         role: 'user' | 'assistant' | 'system'
         content: string
         timestamp: string
         isStreaming?: boolean  // true while assistant is actively responding
       }

       export const useChatStore = defineStore('chat', () => {
         const messages = ref<ChatMessage[]>([])
         const sessionId = ref<string>(crypto.randomUUID())
         const isOpen = ref(false)  // chat panel open/closed state
         const isLoading = ref(false)  // waiting for AI response
         const toolMessage = ref<string | null>(null)  // current tool progress message

         const lastMessages = computed(() => messages.value.slice(-20))

         function addUserMessage(content: string) {
           messages.value.push({
             id: crypto.randomUUID(),
             role: 'user',
             content,
             timestamp: new Date().toISOString(),
           })
         }

         function addAssistantMessage(content: string, isStreaming = true) {
           const msg: ChatMessage = {
             id: crypto.randomUUID(),
             role: 'assistant',
             content,
             timestamp: new Date().toISOString(),
             isStreaming,
           }
           messages.value.push(msg)
           return msg
         }

         function updateLastAssistantMessage(content: string) {
           const last = messages.value.filter(m => m.role === 'assistant').pop()
           if (last) {
             last.content = content
             last.isStreaming = false
           }
         }

         function setToolMessage(msg: string | null) {
           toolMessage.value = msg
         }

         function toggleOpen() {
           isOpen.value = !isOpen.value
         }

         function clearSession() {
           messages.value = []
           sessionId.value = crypto.randomUUID()
           toolMessage.value = null
         }

         return {
           messages, sessionId, isOpen, isLoading, toolMessage,
           lastMessages,
           addUserMessage, addAssistantMessage, updateLastAssistantMessage,
           setToolMessage, toggleOpen, clearSession,
         }
       })
       ```

    2. Create `src/frontend/src/composables/useChat.ts` — chat SSE composable:
       - Import `useSSE` from existing composable
       - Import `useChatStore` from the new store
       - Function signature: `export function useChat()`
       - On `sendMessage(message: string)`:
         a. Call `store.addUserMessage(message)`
         b. Set `store.isLoading = true`
         c. Use `useSSE('/api/chat', { message, session_id: store.sessionId })` to create SSE connection
         d. Call `sse.start()`
         e. Watch `sse.events` — on each new event:
            - If `event_type === 'tool_executing'`: set `store.toolMessage = event.message`
            - If `event_type === 'tool_completed'`: clear `store.toolMessage`
            - If `event_type === 'chat_response'`: call `store.addAssistantMessage(event.message)` or `store.updateLastAssistantMessage(event.message)`
            - If `event_type === 'done'`: set `store.isLoading = false`
            - If `event_type === 'error'`: set `store.isLoading = false`, show error in chat
         f. Watch `sse.done` — when true, set `store.isLoading = false`
       - Return `{ sendMessage, abort: sse.abort }`
       - Use the existing `useSSE` interface as-is — it already handles POST, parsing, and termination
       - The composable wraps useSSE and connects it to the Pinia store

    3. Register the store in `src/frontend/src/stores/index.ts` if needed (check current pattern).

    Note: The existing `useSSE` creates a new connection per `start()` call. Since each chat message triggers a new SSE connection, this works naturally — each `sendMessage` creates a fresh `useSSE` instance.

    import { ref, nextTick, watch } from 'vue'
    import { NButton, NInput, NIcon, NScrollbar, NSpin } from 'naive-ui'
    import { useChatStore } from '@/stores/chat'
    import { useChat } from '@/composables/useChat'

    const store = useChatStore()
    const { sendMessage } = useChat()
    const inputText = ref('')
    const messageListRef = ref<HTMLElement | null>(null)

    function handleSend() {
      const msg = inputText.value.trim()
      if (!msg || store.isLoading) return
      inputText.value = ''
      sendMessage(msg)
    }

    // Auto-scroll to bottom when new messages arrive
    watch(() => store.messages.length, async () => {

      messageListRef.value?.scrollTo({ top: messageListRef.value.scrollHeight, behavior: 'smooth' })
    })
    </script>
