```python
async def _seed_if_empty():
    import logging
    from pathlib import Path
    from backend.db.seed_pois import seed_pois_from_json

    logger = logging.getLogger(__name__)
    pois_dir = Path("data/pois")

    async with _db_mod.AsyncSessionFactory() as db:
        poi_count = await db.scalar(select(func.count(POI.id)))
        if poi_count == 0:
            # Auto-discover all city POI files
            for poi_file in sorted(pois_dir.glob("*.json")):
                # Derive city name from filename: shanghai.json -> 上海
                # Use a mapping or read from city config
                city_name = _filename_to_city(poi_file.stem)
                if city_name:
                    try:
                        await seed_pois_from_json(str(poi_file), city_name)
                        logger.info(f"Seeded POIs from {poi_file.name} for {city_name}")
                    except Exception as e:
                        logger.error(f"Failed to seed {poi_file.name}: {e}")

        scenario_count = await db.scalar(select(func.count(Scenario.id)))
        if scenario_count == 0:
            with open("data/scenarios/scenarios.json", encoding="utf-8") as f:
                scenarios_data = json.load(f)
            for s in scenarios_data:
                db.add(Scenario(...))
            await db.commit()
            logger.info(f"Seeded {len(scenarios_data)} scenarios")


def _filename_to_city(filename_stem: str) -> str | None:
    """Map POI filename stem to city name."""
    mapping = {
        "shanghai": "上海",
        "hangzhou": "杭州",
    }
    return mapping.get(filename_stem)
```

Key design decisions:
- Use `sorted(glob("*.json"))` for deterministic seeding order
- `_filename_to_city` mapping is a simple dict — extensible for future cities
- Each POI file seeds independently with try/except — one failure doesn't block others
- Scenarios file already contains all cities' scenarios, so no change needed there
- Add logging for observability (which files were seeded, how many POIs)

Import additions: `from pathlib import Path` (add to existing imports or inline)
