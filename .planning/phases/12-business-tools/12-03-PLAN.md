```python
"""Web fetch tool — reads URL content as text (TOOL-02)."""

import ipaddress
from urllib.parse import urlparse

import httpx
from agents import function_tool, RunContextWrapper
from backend.agent.context import AgentContext


def _is_private_ip(ip_str: str) -> bool:
    """Check if IP is private/reserved (SSRF protection per D-18)."""
    try:
        ip = ipaddress.ip_address(ip_str)
        return ip.is_private or ip.is_loopback or ip.is_reserved
    except ValueError:
        return True  # Invalid IP = block

async def _validate_url(url: str) -> None:
    """Resolve hostname and reject private IPs."""
    parsed = urlparse(url)
    if parsed.scheme not in ("http", "https"):
        raise ValueError(f"Unsupported scheme: {parsed.scheme}")
    import socket
    hostname = parsed.hostname
    if not hostname:
        raise ValueError("No hostname in URL")
    # Resolve and check
    addr_infos = socket.getaddrinfo(hostname, None)
    for family, type_, proto, canonname, sockaddr in addr_infos:
        ip_str = sockaddr[0]
        if _is_private_ip(ip_str):
            raise ValueError(f"Blocked: hostname resolves to private IP {ip_str}")

@function_tool
async def web_fetch(
    ctx: RunContextWrapper[AgentContext],
    url: str,
) -> str:
    """读取网页内容，获取指定URL的文本信息。
    
    Args:
        url: 要读取的网页URL
    """
```
Implementation:
- Validate URL scheme (http/https only)
- SSRF check: resolve hostname, reject private IPs (10.x, 172.16-31.x, 192.168.x, 127.x, ::1)
- Use httpx.AsyncClient to fetch with 15s timeout
- Extract text from HTML: strip tags with simple regex `re.sub(r'<[^>]+>', ' ', content)` (no need for BeautifulSoup for MVP)
- Decode as UTF-8 (fallback to latin-1)
- Truncate to 3000 chars per D-17
- Include source URL in output: `来源: {url}\n\n{content}`
- Handle errors: network timeout, DNS failure, HTTP errors → return error message string
