   ```python
   from backend.db.seed_pois import seed_pois_from_json
   import json
   import uuid

   async def _seed_if_empty():
       """Auto-seed POIs and scenarios on fresh DB (D-20: idempotent)."""
       from backend.db.init_db import AsyncSessionFactory
       async with AsyncSessionFactory() as db:
           # Seed POIs
           poi_count = await db.scalar(select(func.count(POI.id)))
           if poi_count == 0:
               await seed_pois_from_json("data/pois/shanghai.json", "上海")
           
           # Seed scenarios
           scenario_count = await db.scalar(select(func.count(Scenario.id)))
           if scenario_count == 0:
               with open("data/scenarios/scenarios.json", encoding="utf-8") as f:
                   scenarios_data = json.load(f)
               for s in scenarios_data:
                   db.add(Scenario(
                       id=s["id"],
                       name=s["name"],
                       description=s["description"],
                       user_input=s["user_input"],
                       city=s["city"],
                       tags=json.dumps(s.get("tags", []), ensure_ascii=False),
                   ))
               await db.commit()

   await _seed_if_empty()
   ```
